        let currentQuestion = 0;
        let score = 0;
        let answered = false;

        const startScreen = document.getElementById('start-screen');
        const quizScreen = document.getElementById('quiz-screen');
        const resultScreen = document.getElementById('result-screen');
        const questionCounter = document.getElementById('question-counter');
        const scoreTracker = document.getElementById('score-tracker');
        const progressBar = document.getElementById('progress-bar');
        const questionText = document.getElementById('question-text');
        const optionsContainer = document.getElementById('options-container');
        const feedbackContainer = document.getElementById('feedback-container');
        const nextBtn = document.getElementById('next-btn');

        function startQuiz() {
            startScreen.classList.add('hidden');
            quizScreen.classList.remove('hidden');
            currentQuestion = 0;
            score = 0;
            loadQuestion();
        }

        function loadQuestion() {
            answered = false;
            feedbackContainer.classList.add('hidden');
            nextBtn.classList.add('hidden');
            
            const q = quizData[currentQuestion];
            questionCounter.textContent = `Pregunta ${currentQuestion + 1} de ${quizData.length}`;
            scoreTracker.textContent = `Puntuación: ${score}`;
            progressBar.style.width = `${((currentQuestion + 1) / quizData.length) * 100}%`;
            questionText.textContent = q.question;

            optionsContainer.innerHTML = '';
            q.options.forEach((opt, index) => {
                const btn = document.createElement('button');
                btn.className = 'w-full text-left p-4 rounded-xl border border-gray-200 hover:bg-indigo-50 hover:border-indigo-300 transition duration-200 text-gray-700 font-medium';
                btn.textContent = opt;
                btn.onclick = () => selectOption(index, btn);
                optionsContainer.appendChild(btn);
            });
        }

        function selectOption(selectedIndex, btnElement) {
            if (answered) return;
            answered = true;

            const q = quizData[currentQuestion];
            const buttons = optionsContainer.children;

            if (selectedIndex === q.correct) {
                btnElement.classList.remove('border-gray-200', 'hover:bg-indigo-50');
                btnElement.classList.add('bg-green-100', 'border-green-500', 'text-green-800');
                score += 10;
                feedbackContainer.className = 'p-4 rounded-xl mb-6 text-sm bg-green-50 text-green-800 border border-green-200';
                feedbackContainer.innerHTML = `<strong>¡Correcto!</strong> ${q.explanation}`;
            } else {
                btnElement.classList.remove('border-gray-200', 'hover:bg-indigo-50');
                btnElement.classList.add('bg-red-100', 'border-red-500', 'text-red-800');
                buttons[q.correct].classList.remove('border-gray-200', 'hover:bg-indigo-50');
                buttons[q.correct].classList.add('bg-green-100', 'border-green-500', 'text-green-800');
                feedbackContainer.className = 'p-4 rounded-xl mb-6 text-sm bg-red-50 text-red-800 border border-red-200';
                feedbackContainer.innerHTML = `<strong>Incorrecto.</strong> ${q.explanation}`;
            }

            scoreTracker.textContent = `Puntuación: ${score}`;
            feedbackContainer.classList.remove('hidden');
            nextBtn.classList.remove('hidden');

            // Deshabilitar todos los botones
            for (let b of buttons) {
                b.disabled = true;
            }
        }

        function nextQuestion() {
            currentQuestion++;
            if (currentQuestion < quizData.length) {
                loadQuestion();
            } else {
                showResults();
            }
        }

        function showResults() {
            quizScreen.classList.add('hidden');
            resultScreen.classList.remove('hidden');

            const maxScore = quizData.length * 10;
            const percentage = (score / maxScore) * 100;
            
            const resultMessage = document.getElementById('result-message');
            const resultDetails = document.getElementById('result-details');

            resultDetails.textContent = `Has obtenido ${score} puntos de ${maxScore} posibles (${percentage.toFixed(0)}%).`;

            if (percentage >= 80) {
                resultMessage.textContent = "🎉 ¡Excelente trabajo! Tienes los conceptos súper claros para el parcial.";
                resultMessage.className = "text-xl font-bold text-green-600 my-4";
            } else if (percentage >= 50) {
                resultMessage.textContent = "👍 ¡Buen intento! Repasa algunos factores teóricos y estarás listo.";
                resultMessage.className = "text-xl font-bold text-yellow-600 my-4";
            } else {
                resultMessage.textContent = "📚 Te sugerimos repasar los textos y apuntes antes de rendir el lunes.";
                resultMessage.className = "text-xl font-bold text-red-600 my-4";
            }
        }

        function restartQuiz() {
            resultScreen.classList.add('hidden');
            startScreen.classList.remove('hidden');
        }
